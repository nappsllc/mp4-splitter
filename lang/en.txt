# MP4 Splitter UI strings: English, the source language and the template for
# translations (lang/<tag>.txt, same keys). Format: `key = value`, one per line;
# {name} placeholders must stay as they are; plural strings have one line per
# CLDR form the language uses: key[zero|one|two|few|many|other]. Metadata:
# @name (the language in itself), @dir (ltr or rtl), @plural (the rule: see
# src/plural.rs). Keys in parentheses, such as (Ctrl+E) or (F1), stay as written.
@name = English
@dir = ltr
@plural = one_other
